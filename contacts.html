import os from http.server import BaseHTTPRequestHandler, HTTPServer import urllib.parse # Автоматически определяем путь к корню проекта (на уровень выше, чем папка src) BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__))) HTML_FILE_PATH = os.path.join(BASE_DIR, "contacts.html") class MyWebServer(BaseHTTPRequestHandler): # Обработка GET-запросов def do_GET(self): if self.path in ['/', '/contacts', '/index.html']: self.send_response(200) self.send_header("Content-type", "text/html; charset=utf-8") self.end_headers() # Чтение файла с помощью контекстного менеджера по выверенному пути try: with open(HTML_FILE_PATH, "r", encoding="utf-8") as file: html_content = file.read() self.wfile.write(bytes(html_content, "utf-8")) except FileNotFoundError: self.wfile.write(bytes("

Ошибка: Файл contacts.html не найден!

", "utf-8")) # Обработка 404 ошибки else: self.send_response(404) self.send_header("Content-type", "text/html; charset=utf-8") self.end_headers() error_html = """ 404 - Страница не найдена

404

Ой! Страница не найдена.

Запрашиваемый вами адрес не существует.

Вернуться на Главную
""" self.wfile.write(bytes(error_html, "utf-8")) # Обработка POST-запросов (вывод данных формы в консоль) def do_POST(self): content_length = int(self.headers['Content-Length']) post_data = self.rfile.read(content_length).decode('utf-8') parsed_data = urllib.parse.parse_qs(post_data) print("\n=== ПОЛУЧЕНЫ ДАННЫЕ ОТ ПОЛЬЗОВАТЕЛЯ ===") for key, value in parsed_data.items(): print(f"{key}: {value if value else ''}") print("========================================\n") self.send_response(200) self.send_header("Content-type", "text/html; charset=utf-8") self.end_headers() success_html = """ Успешно

Данные успешно получены сервером!

Проверить вывод можно в консоли PyCharm.

Назад к форме
""" self.wfile.write(bytes(success_html, "utf-8")) def run(server_class=HTTPServer, handler_class=MyWebServer, port=8000): server_address = ('', port) httpd = server_class(server_address, handler_class) print(f"Сервер успешно запущен на http://localhost:{port}") try: httpd.serve_forever() except KeyboardInterrupt: print("\nСервер остановлен.") if __name__ == '__main__': run()
